	if isfile and readfile and isfile(CONFIG_FILE) then
		pcall(function()
			local decoded = HttpService:JSONDecode(readfile(CONFIG_FILE))
			if type(decoded) == "table" then

-- TÍNH NĂNG NOCLIP: LƯU VÀ KHÔI PHỤC TRẠNG THÁI VA CHẠM GỐC
local jumpReqConn, tpTool, shiftLockGui, shiftLockRender
local noclipConn = nil
local originalCollision = {}

local function restoreCollision()
	for part, oldValue in pairs(originalCollision) do
		if part and part.Parent then
			part.CanCollide = oldValue
		end
	end
	table.clear(originalCollision)
end

local function applyNoclipToCharacter(char)
	if not char then return end

	for _, part in ipairs(char:GetDescendants()) do
		if part:IsA("BasePart") then
			-- Lưu trạng thái gốc đúng một lần, kể cả khi giá trị gốc là false.
			if originalCollision[part] == nil then
				originalCollision[part] = part.CanCollide
			end
			part.CanCollide = false
		end
	end
end

local function toggleNoclip(btn, state)
	if state ~= nil then
		userConfig.Noclip = state
	else
		userConfig.Noclip = not userConfig.Noclip
	end

	if userConfig.Noclip then
		if noclipConn then
			noclipConn:Disconnect()
			noclipConn = nil
		end

		local char = LocalPlayer.Character
		if char then
			applyNoclipToCharacter(char)
		end

		noclipConn = RunService.Stepped:Connect(function()
			if userConfig.Noclip then
				local currentChar = LocalPlayer.Character
				if currentChar then
					applyNoclipToCharacter(currentChar)
				end
			end
		end)
	else
		if noclipConn then
			noclipConn:Disconnect()
			noclipConn = nil
		end

		-- Khôi phục CanCollide theo từng part, không ép tất cả thành true.
		restoreCollision()
	end

	saveConfig()
end

-- Khi nhân vật hồi sinh, bỏ trạng thái của nhân vật cũ và áp dụng lại
-- Noclip cho nhân vật mới nếu tính năng vẫn đang bật.
LocalPlayer.CharacterAdded:Connect(function(char)
	restoreCollision()

	task.defer(function()
		if userConfig.Noclip and LocalPlayer.Character == char then
			applyNoclipToCharacter(char)
		end
	end)
end)

-- X-RAY: làm các BasePart trong map bán trong suốt ở phía client và khôi phục khi tắt.
local xrayOriginal = {}
local xrayConn = nil
local xrayTransparency = 0.65
local function applyXrayToPart(part)
    if not part:IsA("BasePart") then return end
    if LocalPlayer.Character and part:IsDescendantOf(LocalPlayer.Character) then return end
    if xrayOriginal[part] == nil then xrayOriginal[part] = part.LocalTransparencyModifier end
    part.LocalTransparencyModifier = xrayTransparency
end
local function toggleXray(btn, state)
    if state ~= nil then userConfig.XRay = state else userConfig.XRay = not userConfig.XRay end
    if userConfig.XRay then
        for _, obj in ipairs(Workspace:GetDescendants()) do applyXrayToPart(obj) end
        if xrayConn then xrayConn:Disconnect() end
        xrayConn = Workspace.DescendantAdded:Connect(function(obj)
            if userConfig.XRay then applyXrayToPart(obj) end
        end)
    else
        if xrayConn then xrayConn:Disconnect(); xrayConn = nil end
        for part, old in pairs(xrayOriginal) do
            if part and part.Parent then part.LocalTransparencyModifier = old end
        end
        table.clear(xrayOriginal)
    end
    saveConfig()
end

-- SPEED: người dùng nhập WalkSpeed mong muốn; bật/tắt riêng.
local speedEnabled = false
local speedOriginal = nil
local speedValue = 32
local speedConn = nil
local function applySpeed()
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum and speedEnabled then hum.WalkSpeed = speedValue end
end
local function toggleSpeed(btn, state)
    if state ~= nil then speedEnabled = state else speedEnabled = not speedEnabled end
    userConfig.Speed = speedEnabled
    if speedEnabled then
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then speedOriginal = hum.WalkSpeed end
        applySpeed()
        if speedConn then speedConn:Disconnect() end
        speedConn = LocalPlayer.CharacterAdded:Connect(function(char)
            char:WaitForChild("Humanoid", 5)
            task.wait(0.1)
            applySpeed()
        end)
    else
        if speedConn then speedConn:Disconnect(); speedConn = nil end
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum and speedOriginal then hum.WalkSpeed = speedOriginal end
        speedOriginal = nil
    end
    saveConfig()
end

-- Inf Zoom độc lập: tăng giới hạn zoom và dùng Invisicam để camera làm mờ vật cản.
local oldCameraMaxZoom = nil
local oldCameraOcclusion = nil
local function toggleInfZoom(btn, state)
	if state ~= nil then userConfig.InfZoom = state else userConfig.InfZoom = not userConfig.InfZoom end
	if userConfig.InfZoom then
		if oldCameraMaxZoom == nil then oldCameraMaxZoom = LocalPlayer.CameraMaxZoomDistance end
		if oldCameraOcclusion == nil then oldCameraOcclusion = LocalPlayer.DevCameraOcclusionMode end
		LocalPlayer.CameraMaxZoomDistance = 100000
		pcall(function() LocalPlayer.DevCameraOcclusionMode = Enum.DevCameraOcclusionMode.Invisicam end)
	else
		LocalPlayer.CameraMaxZoomDistance = oldCameraMaxZoom or 128
		if oldCameraOcclusion ~= nil then pcall(function() LocalPlayer.DevCameraOcclusionMode = oldCameraOcclusion end) end
		oldCameraMaxZoom, oldCameraOcclusion = nil, nil
	end
	saveConfig()
end

local function toggleShiftLock(btn, state)
	if state ~= nil then userConfig.ShiftLock = state else userConfig.ShiftLock = not userConfig.ShiftLock end
	if userConfig.ShiftLock then
		pcall(function() LocalPlayer.DevEnableMouseLock = true end)
		if not shiftLockGui then
			shiftLockGui = Instance.new("ScreenGui") shiftLockGui.Name = "zestShiftLockUI" shiftLockGui.ResetOnSpawn = false
			pcall(function() shiftLockGui.Parent = CoreGui end)
			if not shiftLockGui.Parent then shiftLockGui.Parent = LocalPlayer:WaitForChild("PlayerGui") end
			local slBtn = Instance.new("ImageButton")
			slBtn.Size = UDim2.new(0, 50, 0, 50) slBtn.Position = UDim2.new(1, -120, 0.5, -25)
			slBtn.Image = "rbxasset://textures/ui/mouseLock_off.png" slBtn.BackgroundTransparency = 1 slBtn.Parent = shiftLockGui
			local isLocked = false
			slBtn.MouseButton1Click:Connect(function()
				isLocked = not isLocked
				if isLocked then
					slBtn.Image = "rbxasset://textures/ui/mouseLock_on.png"
					UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter
					shiftLockRender = RunService.RenderStepped:Connect(function()
						local char = LocalPlayer.Character
						local cam = workspace.CurrentCamera
						if char and char:FindFirstChild("HumanoidRootPart") and char:FindFirstChildOfClass("Humanoid") then
							char:FindFirstChildOfClass("Humanoid").AutoRotate = false
							local look = cam.CFrame.lookVector
							local root = char.HumanoidRootPart
							root.CFrame = CFrame.new(root.Position, root.Position + Vector3.new(look.X, 0, look.Z))
						end
					end)
				else
					slBtn.Image = "rbxasset://textures/ui/mouseLock_off.png"
					UserInputService.MouseBehavior = Enum.MouseBehavior.Default
					if shiftLockRender then shiftLockRender:Disconnect() shiftLockRender = nil end
					if LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid") then
						LocalPlayer.Character:FindFirstChildOfClass("Humanoid").AutoRotate = true
					end
				end
			end)
		end
	else
		if shiftLockGui then shiftLockGui:Destroy() shiftLockGui = nil end
		if shiftLockRender then shiftLockRender:Disconnect() shiftLockRender = nil end
		UserInputService.MouseBehavior = Enum.MouseBehavior.Default
	end
	saveConfig()
end

-- FIX LAG NÂNG CẤP: 4 mức, lưu trạng thái gốc để khôi phục khi tắt.
local fixLagSnapshot = {parts = {}, visuals = {}}
local fixLagAddedConn
local oldRenderQuality, oldGlobalShadows, oldFogEnd
local fixLagProfiles = {
	[1] = {Name="NHẸ", Quality=Enum.QualityLevel.Automatic, HideTextures=false, DisableEffects=false, Simplify=false},
	[2] = {Name="VỪA", Quality=Enum.QualityLevel.Level05, HideTextures=false, DisableEffects=true, Simplify=false},
	[3] = {Name="MẠNH", Quality=Enum.QualityLevel.Level03, HideTextures=true, DisableEffects=true, Simplify=true},
	[4] = {Name="CỰC MẠNH", Quality=Enum.QualityLevel.Level01, HideTextures=true, DisableEffects=true, Simplify=true}
}
local function optimizeObject(obj)
	local profile = fixLagProfiles[math.clamp(tonumber(userConfig.FixLagLevel) or 2, 1, 4)]
	if obj:IsA("BasePart") then
		if not fixLagSnapshot.parts[obj] then
			fixLagSnapshot.parts[obj] = {Material=obj.Material, Reflectance=obj.Reflectance, CastShadow=obj.CastShadow}
		end
		obj.CastShadow = false
		if profile.Simplify then obj.Material = Enum.Material.SmoothPlastic; obj.Reflectance = 0 end
	elseif obj:IsA("Decal") or obj:IsA("Texture") then
		if profile.HideTextures then
			if fixLagSnapshot.visuals[obj] == nil then fixLagSnapshot.visuals[obj] = {Kind="Transparency", Value=obj.Transparency} end
			obj.Transparency = 1
		end
	elseif profile.DisableEffects and (obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") or obj:IsA("Smoke") or obj:IsA("Fire") or obj:IsA("Sparkles")) then
		if fixLagSnapshot.visuals[obj] == nil then fixLagSnapshot.visuals[obj] = {Kind="Enabled", Value=obj.Enabled} end
		obj.Enabled = false
	end
end
local function applyFixLag()
	if oldGlobalShadows == nil then
		oldGlobalShadows, oldFogEnd = Lighting.GlobalShadows, Lighting.FogEnd
		pcall(function() oldRenderQuality = settings().Rendering.QualityLevel end)
	end
	for _, obj in ipairs(Workspace:GetDescendants()) do pcall(optimizeObject, obj) end
	local profile = fixLagProfiles[math.clamp(tonumber(userConfig.FixLagLevel) or 2, 1, 4)]
	Lighting.GlobalShadows = false
	if profile.HideTextures then Lighting.FogEnd = 100000 end
	pcall(function() settings().Rendering.QualityLevel = profile.Quality end)
	if fixLagAddedConn then fixLagAddedConn:Disconnect() end
	fixLagAddedConn = Workspace.DescendantAdded:Connect(function(obj)
		if userConfig.FixLag then task.defer(function() if obj and obj.Parent then pcall(optimizeObject, obj) end end) end
	end)
end
local function restoreFixLag()
	if fixLagAddedConn then fixLagAddedConn:Disconnect() fixLagAddedConn = nil end
	for obj, old in pairs(fixLagSnapshot.parts) do
		if obj and obj.Parent then pcall(function() obj.Material=old.Material; obj.Reflectance=old.Reflectance; obj.CastShadow=old.CastShadow end) end
	end
	for obj, old in pairs(fixLagSnapshot.visuals) do
		if obj and obj.Parent then pcall(function() if old.Kind == "Transparency" then obj.Transparency=old.Value else obj.Enabled=old.Value end end) end
	end
	table.clear(fixLagSnapshot.parts); table.clear(fixLagSnapshot.visuals)
	if oldGlobalShadows ~= nil then Lighting.GlobalShadows = oldGlobalShadows end
	if oldFogEnd ~= nil then Lighting.FogEnd = oldFogEnd end
	if oldRenderQuality ~= nil then pcall(function() settings().Rendering.QualityLevel = oldRenderQuality end) end
end
local function toggleFixLag(btn, state)
	if state ~= nil then userConfig.FixLag = state else userConfig.FixLag = not userConfig.FixLag end
	if userConfig.FixLag then applyFixLag() else restoreFixLag() end
	saveConfig()
end
local function setFixLagLevel(level)
	userConfig.FixLagLevel = level
	if userConfig.FixLag then
		-- Khôi phục bản cũ trước khi áp dụng mức mới, giữ nguyên snapshot gốc.
		if fixLagAddedConn then fixLagAddedConn:Disconnect(); fixLagAddedConn = nil end
		for obj, old in pairs(fixLagSnapshot.parts) do
			if obj and obj.Parent then pcall(function() obj.Material=old.Material; obj.Reflectance=old.Reflectance; obj.CastShadow=old.CastShadow end) end
		end
		for obj, old in pairs(fixLagSnapshot.visuals) do
			if obj and obj.Parent then pcall(function() if old.Kind == "Transparency" then obj.Transparency=old.Value else obj.Enabled=old.Value end end) end
		end
		-- Xóa snapshot tạm, chụp lại trạng thái hiện tại sau khi phục hồi.
		table.clear(fixLagSnapshot.parts); table.clear(fixLagSnapshot.visuals)
		applyFixLag()
	end
	saveConfig()
end

-- ESP Player nền cầu vồng (giữ lại, không kèm Aimbot).
local function clearESP()
	for _, obj in pairs(espObjects) do if obj and obj.Parent then obj:Destroy() end end
	table.clear(espObjects)
end
local function addESP(player)
	if player == LocalPlayer or not userConfig.ESPPlayer then return end
	local char = player.Character
	if not char then return end
	local old = espObjects[player]
	if old then old:Destroy() end
	local highlight = Instance.new("Highlight")
	highlight.Name = "zestPlayerESP"
	local c = Color3.fromHSV(espRainbowClock % 1, 0.9, 1)
	highlight.FillColor = c
	highlight.OutlineColor = c
	highlight.FillTransparency = 0.65
	highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	highlight.Adornee = char
	highlight.Parent = char
	espObjects[player] = highlight
end
local function toggleESPPlayer(btn, state)
	if state ~= nil then userConfig.ESPPlayer = state else userConfig.ESPPlayer = not userConfig.ESPPlayer end
	if espRainbowConn then espRainbowConn:Disconnect(); espRainbowConn = nil end
	clearESP()
	if userConfig.ESPPlayer then
		for _, player in ipairs(Players:GetPlayers()) do addESP(player) end
		espRainbowConn = RunService.Heartbeat:Connect(function(dt)
			espRainbowClock = (espRainbowClock + dt * 0.16) % 1
			local c = Color3.fromHSV(espRainbowClock, 0.9, 1)
			for player, highlight in pairs(espObjects) do
				if highlight and highlight.Parent then highlight.FillColor = c; highlight.OutlineColor = c else espObjects[player] = nil end
			end
		end)
	end
	saveConfig()
end
Players.PlayerAdded:Connect(function(player)
	player.CharacterAdded:Connect(function() task.wait(0.5); addESP(player) end)
end)
for _, player in ipairs(Players:GetPlayers()) do
	player.CharacterAdded:Connect(function() task.wait(0.5); addESP(player) end)
end
Players.PlayerRemoving:Connect(function(player)
	if espObjects[player] then espObjects[player]:Destroy(); espObjects[player] = nil end
end)
